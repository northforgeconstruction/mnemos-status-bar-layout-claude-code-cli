```python
def validate_batch_judges(judges: list) -> None:
    bad = [j for j in judges if JUDGE_MODELS[j][0] != "anthropic"]
    if bad:
        print(f"--batch supports anthropic judges only; remove: {','.join(bad)}",
              file=sys.stderr)
        raise SystemExit(2)


def build_batch_requests(prompt_template: str, pairs: list, transcripts_dir,
                         judges: list, seed: int):
    """Build Batches API request dicts + a custom_id -> metadata map."""
    requests, meta = [], {}
    seed_counter = seed
    for pair in pairs:
        tx_a = (transcripts_dir / pair["tx_a"]).read_text()
        tx_b = (transcripts_dir / pair["tx_b"]).read_text()
        for judge_id in judges:
            _, model = JUDGE_MODELS[judge_id]
            for dim_code in DIMENSIONS:
                seed_counter += 1
                swap = random.Random(seed_counter).choice([True, False])
                a, b = (tx_b, tx_a) if swap else (tx_a, tx_b)
                prompt = build_query(prompt_template, pair["task_description"],
                                     a, b, DIMENSIONS[dim_code])
                custom_id = re.sub(r"[^A-Za-z0-9_-]", "-",
                                   f"{pair['pair_id']}_{judge_id}_{dim_code}_{seed_counter}")
                requests.append({
                    "custom_id": custom_id,
                    "params": {"model": model, "max_tokens": 1024,
                               "messages": [{"role": "user", "content": prompt}]},
                })
                meta[custom_id] = {"judge": judge_id, "dimension": dim_code,
                                   "swap": swap, "pair_id": pair["pair_id"],
                                   "task_id": pair["task_id"]}
    return requests, meta


def batch_result_to_record(parsed: dict, m: dict) -> dict:
    pref = parsed["preference"]
    if m["swap"] and pref in ("A", "B"):
        pref = "B" if pref == "A" else "A"
    return {"judge": m["judge"], "dimension": m["dimension"], "preference": pref,
            "confidence": parsed.get("confidence", 0),
            "reasoning": parsed.get("reasoning", ""),
            "position_swapped": m["swap"],
            "pair_id": m["pair_id"], "task_id": m["task_id"]}


def run_batch(requests: list, meta: dict, poll_seconds: int = 60) -> list:
    if Anthropic is None:
        raise RuntimeError("anthropic SDK not installed")
    client = Anthropic()
    batch = client.messages.batches.create(requests=requests)
    print(f"[batch] created {batch.id} ({len(requests)} requests)", file=sys.stderr)
    while True:
        batch = client.messages.batches.retrieve(batch.id)
        if batch.processing_status == "ended":
            break
        print(f"[batch] {batch.processing_status} … polling in {poll_seconds}s",
              file=sys.stderr)
        time.sleep(poll_seconds)
    records = []
    for result in client.messages.batches.results(batch.id):
        m = meta.get(result.custom_id)
        if m is None:
            continue
        if result.result.type != "succeeded":
            print(f"[ERR] {result.custom_id}: {result.result.type}", file=sys.stderr)
            continue
        msg = result.result.message
        text = next((b.text for b in msg.content if b.type == "text"), "")
        try:
            records.append(batch_result_to_record(parse_json_response(text), m))
        except Exception as e:
            print(f"[ERR] {result.custom_id}: {e}", file=sys.stderr)
    return records
```
